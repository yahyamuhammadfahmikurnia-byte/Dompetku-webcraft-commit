import { useCallback, useEffect, useState } from "react";
import { onAuthStateChanged } from "firebase/auth";
import Sidebar from "../components/Sidebar";
import SummaryCard from "../components/SummaryCard";
import BudgetChart from "../components/BudgetChart";
import SavingAssistant from "../components/SavingAssistant";
import QuickSimulation from "../components/QuickSimulation";
import { auth } from "../lib/firebase";
import { ambilSimulasi } from "../lib/firestore";
import "./Dashboard.css";

function formatRupiah(value) {
  const angka = Number(value) || 0;
  const hasil = `Rp${Math.abs(angka).toLocaleString("id-ID")}`;

  return angka < 0 ? `-${hasil}` : hasil;
}

function cariSimulasiTerbaru(data) {
  if (!Array.isArray(data) || data.length === 0) {
    return null;
  }

  const urut = [...data].sort((a, b) => {
    const waktuA = Number(a.dibuatPada) || 0;
    const waktuB = Number(b.dibuatPada) || 0;

    if (waktuA !== waktuB) {
      return waktuB - waktuA;
    }

    return String(b.tanggal || "").localeCompare(
      String(a.tanggal || "")
    );
  });

  return urut[0];
}

export default function Dashboard() {
  const [simulasiTerbaru, setSimulasiTerbaru] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [menuTerbuka, setMenuTerbuka] = useState(false);

  const perbaruiData = useCallback(async () => {
    const user = auth.currentUser;

    if (!user) {
      setSimulasiTerbaru(null);
      setLoading(false);
      return;
    }

    setLoading(true);
    setError("");

    try {
      const uid = user.uid;
      const data = await ambilSimulasi();

      if (auth.currentUser?.uid !== uid) {
        return;
      }

      setSimulasiTerbaru(cariSimulasiTerbaru(data));
    } catch (err) {
      if (auth.currentUser?.uid !== user.uid) return;

      console.error("Gagal mengambil data Dashboard:", err);

      setError(
        "Gagal memuat data dari Firebase. Periksa koneksi dan Security Rules."
      );
    } finally {
      if (auth.currentUser?.uid === user.uid) {
        setLoading(false);
      }
    }
  }, []);

  useEffect(() => {
    const unsubscribe = onAuthStateChanged(
      auth,
      (user) => {
        if (user) {
          perbaruiData();
        } else {
          setSimulasiTerbaru(null);
          setLoading(false);
        }
      }
    );

    return () => unsubscribe();
  }, [perbaruiData]);

  useEffect(() => {
    const onEscape = (event) => {
      if (event.key === "Escape") {
        setMenuTerbuka(false);
      }
    };

    window.addEventListener("keydown", onEscape);

    return () => {
      window.removeEventListener("keydown", onEscape);
    };
  }, []);

  const uangBulanan = simulasiTerbaru
    ? Number(simulasiTerbaru.uangBulanan) || 0
    : 0;

  const totalPengeluaran = simulasiTerbaru
    ? Number(simulasiTerbaru.pengeluaran) || 0
    : 0;

  const sisaUang = simulasiTerbaru
    ? Number(simulasiTerbaru.sisa) || 0
    : 0;

  return (
    <div className="dashboard-layout">
      <div className="dashboard-mobile-header">
        <button
          type="button"
          className="dashboard-menu-toggle"
          onClick={() => setMenuTerbuka((prev) => !prev)}
          aria-label={
            menuTerbuka ? "Tutup menu navigasi" : "Buka menu navigasi"
          }
          aria-expanded={menuTerbuka}
          aria-controls="dashboard-sidebar"
        >
          {menuTerbuka ? "✕" : "☰"}
        </button>

        <span className="dashboard-mobile-brand">
          Dompetku
        </span>
      </div>

      {menuTerbuka && (
        <button
          type="button"
          className="dashboard-sidebar-overlay"
          onClick={() => setMenuTerbuka(false)}
          aria-label="Tutup menu"
        />
      )}

      <div
        id="dashboard-sidebar"
        className={`dashboard-sidebar-wrapper ${
          menuTerbuka ? "is-open" : ""
        }`}
      >
        <Sidebar onNavigate={() => setMenuTerbuka(false)} />
      </div>

      <main className="dashboard-content">
        <div className="dashboard-header">
          <div>
            <h1>Halo sobat hemat</h1>
            <p>
              Perhatikan keuanganmu seperti kamu
              memperhatikan dosen kalkulusmu!
            </p>
          </div>

          <span className="simulation-badge">
            Simulasi
          </span>
        </div>

        {error && (
          <p
            role="alert"
            style={{
              padding: "14px 18px",
              marginBottom: "20px",
              background: "#fff1f0",
              color: "#c0392b",
              borderRadius: "10px",
              fontSize: "13px",
            }}
          >
            {error}
          </p>
        )}

        {loading && (
          <p
            role="status"
            style={{
              marginBottom: "16px",
              color: "#147d66",
              fontSize: "13px",
            }}
          >
            Memuat data keuangan dari Firebase...
          </p>
        )}

        {!loading && !error && !simulasiTerbaru && (
          <p
            style={{
              padding: "14px 18px",
              marginBottom: "20px",
              background: "#e6f3ee",
              color: "#147d66",
              borderRadius: "10px",
              fontSize: "13px",
            }}
          >
            Belum ada simulasi tersimpan di Firebase.
            Buka Simulator untuk membuat simulasi pertamamu.
          </p>
        )}

        <div className="summary-grid">
          <SummaryCard
            title="Uang Bulanan"
            amount={formatRupiah(uangBulanan)}
            description="Uang saku dari simulasi terbaru"
            primary
          />

          <SummaryCard
            title="Total Pengeluaran"
            amount={formatRupiah(totalPengeluaran)}
            description="Pengeluaran dari simulasi terbaru"
          />

          <SummaryCard
            title="Sisa Uang"
            amount={formatRupiah(sisaUang)}
            description="Saldo berdasarkan simulasi"
          />
        </div>

        <div className="dashboard-panels">
          <BudgetChart simulasi={simulasiTerbaru} />
          <SavingAssistant simulasi={simulasiTerbaru} />
        </div>

        <QuickSimulation onSimpan={perbaruiData} />
      </main>
    </div>
  );
}

function pesanErrorFirebase(error) {

  // Mengambil profil dari Firebase
  useEffect(() => {
    let aktif = true;
    let requestId = 0;

    const unsubscribe = onAuthStateChanged(
      auth,
      async (user) => {
        const currentRequest = ++requestId;

        if (!user) {
          if (aktif) {
            setLoading(false);
            navigate("/login", { replace: true });
          }
          return;
        }

        setLoading(true);

        const profilFirebase = {
          nama: user.displayName || "",
          email: user.email || "",
          kampus: "",
          status: "Mahasiswa",
        };

        try {
          const data = await ambilProfil();

          if (!aktif || currentRequest !== requestId) {
            return;
          }

          const profilLengkap = {
            nama: data?.nama || profilFirebase.nama,
            email: user.email || "",
            kampus: data?.kampus || "",
            status: data?.status || "Mahasiswa",
          };

          setProfile(profilLengkap);
          setProfileTersimpan(profilLengkap);
          setPesan("");
          setPesanError(false);
        } catch (error) {
          if (!aktif || currentRequest !== requestId) {
            return;
          }

          console.error("Gagal memuat profil:", error);

          setProfile(profilFirebase);
          setProfileTersimpan(profilFirebase);

          setPesan(
            "Profil tambahan belum bisa dimuat dari Firestore. Periksa koneksi dan Security Rules."
          );
          setPesanError(true);
        } finally {
          if (aktif && currentRequest === requestId) {
            setLoading(false);
          }
        }
      },
      (error) => {
        console.error("Gagal memeriksa sesi:", error);

        if (aktif) {
          setLoading(false);
          setPesan("Gagal memeriksa sesi Firebase.");
          setPesanError(true);

function rupiah(angka) {

  const nilai = Number(angka) || 0;

  const hasil = `Rp${Math.abs(nilai).toLocaleString("id-ID")}`;

  return nilai < 0 ? `−${hasil}` : hasil;

function tanggalIndonesia(tanggal) {

  if (!tanggal) return "-";

  const [tahun, bulan, hari] = String(tanggal)

    .split("-")

    .map(Number);

  const date = new Date(tahun, bulan - 1, hari);

  if (Number.isNaN(date.getTime())) return "-";

  return new Intl.DateTimeFormat("id-ID", {

    day: "numeric",

    month: "short",

    year: "numeric",

  }).format(date);

}

function pesanErrorFirestore(error) {

  if (

    error?.code === "permission-denied" ||

    error?.code === "firestore/permission-denied"

  ) {

    return "Akses ditolak. Periksa Security Rules Firestore.";

  }

  if (

    error?.code === "unavailable" ||

    error?.code === "firestore/unavailable"

  ) {

    return "Firebase tidak tersedia. Periksa koneksi internet.";

  }

  return error?.message || "Terjadi kesalahan saat mengakses Firestore.";

}

function tanggalLokal() {
  const sekarang = new Date();
  const tahun = sekarang.getFullYear();
  const bulan = String(sekarang.getMonth() + 1).padStart(2, "0");
  const hari = String(sekarang.getDate()).padStart(2, "0");

  return `${tahun}-${bulan}-${hari}`;
}

function formatRupiah(value) {
  return `Rp${Number(value || 0).toLocaleString("id-ID")}`;
}

function hitungAnggaran(data) {
  const hari = Number(data.periode);
  const uang = Number(data.uangSaku);
  const kos = Number(data.kos);
  const makan = Number(data.makan);
  const transportasi = Number(data.transportasi);
  const lainnya = Number(data.lainnya);

  const kebutuhanMakan = data.hitungMakan
    ? makan * hari
    : 0;

  const pengeluaranTetap = kos + transportasi + lainnya;
  const totalPengeluaran = pengeluaranTetap + kebutuhanMakan;
  const sisa = uang - totalPengeluaran;

  const batasAman =
    hari > 0
      ? Math.max(
          0,
          Math.floor((uang - pengeluaranTetap) / hari)
        )
      : 0;

  return {
    totalPengeluaran,
    sisa,
    makan,
    batasAman,
    potensiTabungan: Math.max(0, sisa),
  };
}

function hitungTarget(data) {

  const target = Math.max(0, Number(data.jumlah) || 0);

  const bulanan = Math.max(0, Number(data.bulanan) || 0);

  const awal = Math.max(0, Number(data.awal) || 0);

  const kekurangan = Math.max(0, target - awal);

  const progres =

    target > 0 ? Math.min(100, (awal / target) * 100) : 0;

  const bulan =

    kekurangan === 0

      ? 0

      : bulanan > 0

        ? Math.ceil(kekurangan / bulanan)

        : null;

  return {

    progres,

    bulan,

    kekurangan,

    terkumpul: awal,

    target,
  };

}

function urutkanTarget(data) {

  return [...data].sort(

  );

}

  const sedangProses = proses !== "";

  const perluHitungUlang =

    JSON.stringify(form) !== JSON.stringify(formTerhitung);

  useEffect(() => {

    let aktif = true;

    let versi = 0;

    const unsubscribe = onAuthStateChanged(

      auth,

      async (user) => {

        const versiSekarang = ++versi;

        setLoading(true);

        setPesan("");

        if (!user) {

          setTargetTersimpan([]);

          setTargetAktif(null);

          setLoading(false);

          return;
        }

        try {

          const data = await ambilTarget();

          if (!aktif || versiSekarang !== versi) return;

          setTargetTersimpan(urutkanTarget(data));
        } catch (error) {

          if (!aktif || versiSekarang !== versi) return;

          setPesan(pesanErrorFirebase(error));

          setPesanError(true);
        } finally {

          if (aktif && versiSekarang === versi) {

            setLoading(false);
          }
        }
      },

      (error) => {

        if (!aktif) return;

        setPesan(pesanErrorFirebase(error));

        setPesanError(true);

        setLoading(false);
      }
    );

    return () => {

      aktif = false;

      versi++;

      unsubscribe();
    };
  }, []);

    const handleEscape = (event) => {
      if (event.key === "Escape") setMenuTerbuka(false);
    };

    window.addEventListener("keydown", handleEscape);
    return () => window.removeEventListener("keydown", handleEscape);
  }, []);

  function tampilkanPesan(teks, error = false) {

    setPesan(teks);

    setPesanError(error);
  }

  function ubahInput(nama, nilai) {

  function handleSubmit(event) {

    event.preventDefault();

    if (!form.nama.trim()) {

      tampilkanPesan("Nama tujuan wajib diisi.", true);

      return;
    }

    if (Number(form.jumlah) <= 0) {

      tampilkanPesan(

        "Jumlah target harus lebih dari Rp0.",

        true
      );

      return;
    }

    const data = {

      ...form,

      nama: form.nama.trim(),
    };

    setForm(data);

    setFormTerhitung(data);

    setHasil(hitungTarget(data));

    setTargetAktif(null);

    tampilkanPesan(

      "Perhitungan selesai. Kamu bisa menyimpan target ini."
    );
  }

  async function handleSimpan() {

    if (proses || loading) return;

    if (targetAktif) {

      setPesan("Target ini sudah tersimpan di Firebase.");

      setPesanError(false);

      return;
    }

    if (!auth.currentUser) {

      setPesan("Silakan login terlebih dahulu.");

      setPesanError(true);

      return;
    }

    const dataBaru = {

      nama: form.nama.trim(),

      jumlah: Number(form.jumlah),

      bulanan: Number(form.bulanan) || 0,

      awal: Number(form.awal) || 0,

      tanggal: new Date().toISOString(),

      dibuatPada: Date.now(),
    };

    if (!dataBaru.nama || dataBaru.jumlah <= 0) {

      setPesan("Isi nama dan jumlah target dengan benar.");

      setPesanError(true);

      return;
    }

    setProses("simpan");

    setPesan("");

    setPesanError(false);

    try {

      const id = await simpanTarget(dataBaru);

      const targetBaru = {

        ...dataBaru,

        id,
      };

      setTargetTersimpan((prev) =>

        urutkanTarget([targetBaru, ...prev])
      );

      const formBaru = {

        nama: dataBaru.nama,

        jumlah: String(dataBaru.jumlah),

        bulanan: String(dataBaru.bulanan),

        awal: String(dataBaru.awal),
      };

      setForm(formBaru);

      setFormTerhitung(formBaru);

      setHasil(hitungTarget(formBaru));

      setTargetAktif(id);

      setTambahan("");

      setPesan(`Target "${dataBaru.nama}" berhasil disimpan.`);

      setPesanError(false);
    } catch (error) {

      console.error("Gagal menyimpan target:", error);

      setPesan(

        `Gagal menyimpan target: ${error.code || error.message}`
      );

      setPesanError(true);
    } finally {

      setProses("");
    }
  }

  function pilihTarget(item) {

    if (sedangProses) return;

    const data = {

      nama: item.nama,

      jumlah: String(item.jumlah),

      bulanan: String(item.bulanan),

      awal: String(item.awal),
    };

    setForm(data);

    setFormTerhitung(data);

    setHasil(hitungTarget(data));

    setTargetAktif(item.id);

    setTambahan("");

    setKonfirmasiHapus(null);

    tampilkanPesan("");
  }

  async function tambahTabungan() {

    if (sedangProses) return;

    if (!targetAktif) {

      tampilkanPesan(

        "Pilih target tersimpan terlebih dahulu.",

        true
      );

      return;

      );

      return;
    }

    const targetDipilih = targetTersimpan.find(

      (item) => item.id === targetAktif
    );

    if (!targetDipilih) {

      tampilkanPesan("Target tidak ditemukan.", true);

      return;
    }

    setProses("tambah");

    tampilkanPesan("");

    try {

      const { increment } = await import("firebase/firestore");

      await perbaruiTarget(targetAktif, {

        awal: increment(nominal),
      });

      const dataTerbaru = await ambilTarget();

      const dataUrut = urutkanTarget(dataTerbaru);

      setTargetTersimpan(dataUrut);

      const targetBaru = dataUrut.find(

        (item) => item.id === targetAktif
      );

      if (targetBaru) {

        const formBaru = {

          nama: targetBaru.nama,

          jumlah: String(targetBaru.jumlah),

          bulanan: String(targetBaru.bulanan),

          awal: String(targetBaru.awal),
        };

        setForm(formBaru);

        true
      );
    } finally {

      setProses("");
    }
  }

  async function hapusTarget(id) {
